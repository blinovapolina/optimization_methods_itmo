ФУНКЦИЯ РаспределитьБригады(n, f1, f2, f3)
    ДОПУСТИМЫЕ_ТОННЕЛЬ ← [0, 2, 4]
    МИН_РАЗВЯЗКА ← 1

    ДЛЯ s ОТ 0 ДО n
        F1[s] ← f1[s]
        X1[s] ← s
    КОНЕЦ ЦИКЛА

    ДЛЯ s ОТ 0 ДО n
        F2[s] ← -∞
        X2[s] ← 0
        ДЛЯ КАЖДОГО x2 ИЗ ДОПУСТИМЫЕ_ТОННЕЛЬ
            ЕСЛИ x2 ≤ s ТОГДА
                значение ← f2[x2] + F1[s - x2]
                ЕСЛИ значение > F2[s] ТОГДА
                    F2[s] ← значение
                    X2[s] ← x2
                КОНЕЦ ЕСЛИ
            КОНЕЦ ЕСЛИ
        КОНЕЦ ЦИКЛА
    КОНЕЦ ЦИКЛА

    F3 ← -∞
    X3 ← 0
    ДЛЯ x3 ОТ МИН_РАЗВЯЗКА ДО n
        значение ← f3[x3] + F2[n - x3]
        ЕСЛИ значение > F3 ТОГДА
            F3 ← значение
            X3 ← x3
        КОНЕЦ ЕСЛИ
    КОНЕЦ ЦИКЛА

    x3 ← X3
    x2 ← X2[n - x3]
    x1 ← n - x3 - x2

    ВЕРНУТЬ (x1, x2, x3, F3)
КОНЕЦ ФУНКЦИИ
